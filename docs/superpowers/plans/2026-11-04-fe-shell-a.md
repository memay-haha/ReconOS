```ts
import { deltaE, nearestToken } from './colour-audit.mjs';

describe('deltaE', () => {
  it('is zero for a colour against itself', () => {
    expect(deltaE('#161d2b', '#161d2b')).toBe(0);
  });

  it('is symmetric', () => {
    expect(deltaE('#94a3b8', '#9aa4b2')).toBeCloseTo(deltaE('#9aa4b2', '#94a3b8'), 10);
  });

  it('calls two greys that differ by one digit indistinguishable', () => {
    // #161d2b is used 118 times and #161d2a 26 times. Below 2.3 the eye
    // cannot tell them apart, so they must collapse to one token.
    expect(deltaE('#161d2b', '#161d2a')).toBeLessThan(2.3);
  });

  it('separates two colours a reader would call different', () => {
    expect(deltaE('#f87171', '#4ade80')).toBeGreaterThan(20);
  });
});

describe('nearestToken', () => {
  const tokens = {
    'fg-muted': '#9aa4b2',
    'border-strong': '#a0a9ba',
    'surface-container-low': '#131c27',
  };

  it('matches within the same role, not merely the closest colour', () => {
    // #94a3b8 is used as text. border-strong is closer in raw distance but
    // is a border token, so a text usage must not land on it.
    const hit = nearestToken('#94a3b8', 'text', tokens);
    expect(hit.token).toBe('fg-muted');
  });

  it('reports the distance so a caller can refuse a loose match', () => {
    const hit = nearestToken('#f87171', 'text', tokens);
    expect(hit.deltaE).toBeGreaterThan(5);
  });
```

Tạo `frontend/scripts/colour-audit.mjs`:

```js
import { readdirSync, readFileSync } from 'node:fs';
import { join } from 'node:path';

/** Which token names may serve which Tailwind property. A colour used as text
 * must not be matched to a border token just because the hex is close. */
const ROLE_OF_TOKEN = {
  text: (n) =>
    n.startsWith('fg') ||
    n.startsWith('on-') ||
    n.startsWith('asset-') ||
    ['primary', 'good', 'warn', 'crit', 'info', 'error', 'accent-ink'].includes(n),
  bg: (n) =>
    n.startsWith('bg') || n.startsWith('surface') || n.endsWith('-soft') || n.endsWith('-container'),
  border: (n) =>
    n.startsWith('border') || n.startsWith('outline') || n.endsWith('-bd') || n === 'ring',
};

function srgbToLab(hex) {
  const h = hex.replace('#', '');
  const channel = (v) => {
    const c = parseInt(v, 16) / 255;
    return c <= 0.04045 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4;
  };
  const r = channel(h.slice(0, 2));
  const g = channel(h.slice(2, 4));
  const b = channel(h.slice(4, 6));
  const x = (0.4124 * r + 0.3576 * g + 0.1805 * b) / 0.95047;
  const y = 0.2126 * r + 0.7152 * g + 0.0722 * b;
  const z = (0.0193 * r + 0.1192 * g + 0.9505 * b) / 1.08883;
  const f = (t) => (t > 0.008856 ? Math.cbrt(t) : 7.787 * t + 16 / 116);
  const [fx, fy, fz] = [f(x), f(y), f(z)];
  return [116 * fy - 16, 500 * (fx - fy), 200 * (fy - fz)];
}

export function deltaE(a, b) {
  const [l1, a1, b1] = srgbToLab(a);
  const [l2, a2, b2] = srgbToLab(b);
  return Math.hypot(l1 - l2, a1 - a2, b1 - b2);
}

export function nearestToken(hex, role, tokens) {
  const fits = ROLE_OF_TOKEN[role] ?? (() => true);
  const names = Object.keys(tokens).filter(fits);
  const pool = names.length > 0 ? names : Object.keys(tokens);
  let best = { token: pool[0], deltaE: Infinity };
  for (const name of pool) {
    const d = deltaE(hex, tokens[name]);
    if (d < best.deltaE) best = { token: name, deltaE: d };
  }
  return best;
}

export function collectMockupColours(dir) {
  const found = new Map();
  for (const file of readdirSync(dir).filter((f) => f.endsWith('.html'))) {
    const text = readFileSync(join(dir, file), 'utf8');
    const re = /\b(bg|text|border|ring)-\[(#[0-9a-fA-F]{6})\]/g;
    for (const m of text.matchAll(re)) {
      const role = m[1] === 'ring' ? 'border' : m[1];
      const hex = m[2].toLowerCase();
      const entry = found.get(hex) ?? { roles: {}, count: 0 };
      entry.roles[role] = (entry.roles[role] ?? 0) + 1;
      entry.count += 1;
      found.set(hex, entry);
    }
  }
  for (const entry of found.values()) {
    entry.role = Object.entries(entry.roles).sort((a, b) => b[1] - a[1])[0][0];
  }
  return found;
}
```

```ts
import { fileURLToPath } from 'node:url';

import { describe, expect, it } from 'vitest';

const srcRoot = fileURLToPath(new URL('../', import.meta.url));

/** Vietnamese letters that no English string contains. Plain ASCII words that
 * happen to be Vietnamese are not worth chasing; the diacritics are. */
const VIETNAMESE = /[ăâđêôơưĂÂĐÊÔƠƯàáảãạằắẳẵặầấẩẫậèéẻẽẹềếểễệìíỉĩịòóỏõọồốổỗộờớởỡợùúủũụừứửữựỳýỷỹỵ]/;

function stripComments(code: string): string {
  return code.replace(/\/\*[\s\S]*?\*\//g, '').replace(/^\s*\/\/.*$/gm, '');
}

describe('user-facing text', () => {
  it('is English everywhere in src', () => {
    const files = globSync('**/*.{ts,tsx}', { cwd: srcRoot })
      .filter((f) => !f.endsWith('no-vietnamese.test.ts'))
      .map((f) => `${srcRoot}${f}`);

    const offenders: string[] = [];
    for (const file of files) {
      const code = stripComments(readFileSync(file, 'utf8'));
      code.split('\n').forEach((line, i) => {
        if (VIETNAMESE.test(line)) offenders.push(`${file.replace(srcRoot, '')}:${i + 1}`);
      });
    }
    expect(offenders).toEqual([]);
  });
});
```

```ts
import type { RequestHandler } from 'msw';

import { allHandlers } from './handlers';

/** Endpoints the backend really serves. A handler is never built for these, so
 * the browser talks to the real API and a mock can never shadow working code.
 * When a batch implements an endpoint, add its line here and delete the
 * handler; the test below is what makes that step impossible to forget. */
export const IMPLEMENTED: readonly string[] = [
  'POST /auth/login',
  'POST /auth/logout',
  'GET /auth/me',
  'POST /auth/password',
  'GET /users',
  'POST /users',
  'GET /users/:user_id',
  'PATCH /users/:user_id',
  'POST /users/:user_id/lock',
  'POST /users/:user_id/unlock',
  'GET /users/:user_id/sessions',
  'GET /user-sessions',
  'DELETE /user-sessions/:user_session_id',
  'GET /audit-logs',
];

export function handlersFor(implemented: readonly string[]): RequestHandler[] {
  const live = new Set(implemented);
  return allHandlers.filter((h) => {
    const method = String(h.info.method).toUpperCase();
    const path = String(h.info.path).replace('/api/v1', '');
    return !live.has(`${method} ${path}`);
  });
}
```

```ts
import { readFileSync } from 'node:fs';

/** Elements that carry no structure, only decoration. Comparing them would
 * make the test break on every icon change. */
const SKIP = new Set(['svg', 'path', 'img', 'br', 'script', 'style']);

/** The structural skeleton of a subtree: tag, nesting depth, sibling order and
 * ARIA role. Classes and text are deliberately left out, so restyling a screen
 * does not fail the test but moving a region does. */
export function shapeOf(node: Element, depth = 0): string[] {
  const lines: string[] = [];
  const tag = node.tagName.toLowerCase();
  if (SKIP.has(tag)) return lines;
  const role = node.getAttribute('role');
  lines.push(`${'  '.repeat(depth)}${tag}${role === null ? '' : `[${role}]`}`);
  for (const child of Array.from(node.children)) {
    lines.push(...shapeOf(child, depth + 1));
  }
  return lines;
}

export function shapeOfMockup(file: string, selector: string, depth = 0): string[] {
  const host = document.createElement('div');
  host.innerHTML = readFileSync(file, 'utf8');
  const root = host.querySelector(selector);
  if (root === null) throw new Error(`No ${selector} in ${file}`);
  return shapeOf(root, depth);
}
```
